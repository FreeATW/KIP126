\section{Open questions and near-126 objects}

% SOURCE: aimpaper/main.tex:164-171, Questions que:2theta6 and que:theta5sq.
\begin{definition}[Order-two $\theta_6$ question]
  \label{def:question-order-two-theta6}
  \uses{def:open-proposition,def:adams-hj,def:adams-detection}
  \notready
  This open proposition asks whether $h_6^2$ detects a Kervaire class
  $\theta_6\in\pi_{126}S^0$ of exact order two.  It is recorded but is not a
  hypothesis of the permanent-cycle proof.
\end{definition}

\begin{definition}[$\theta_5^2=0$ question]
  \label{def:question-theta5-square-zero}
  \uses{def:open-proposition,def:adams-detection}
  \notready

  $\theta_5\in\pi_{62}S^0$ detected by $h_5^2$ such that
  $\theta_5^2=0$.  It is neither assumed nor proved by this project.

  \label{def:near126-conditions}

  propositions, not informal annotations.

\section{Framed Kervaire and coherence contexts}

% SOURCE: aimpaper/main.tex:78-101; PROJECT_BOUNDARY.md, Final geometric conclusions.
\begin{definition}[Framed Kervaire context]
  \label{def:framed-kervaire-context}
  \uses{def:stable-context}
  \lean{KIP126.Kervaire.FramedKervaireContext}
  \lean{KIP126.Kervaire.StableHomotopyData}
  \notready
  For a fixed stable context $\mathcal C$, a framed Kervaire context provides
  closed smooth framed manifolds, framed cobordism, and a Pontryagin--Thom
  correspondence landing in the stable homotopy of the same sphere
  $S^0_{\mathcal C}$.  It also provides the middle-dimensional mod--$2$
  intersection pairing, its quadratic refinement, the Arf invariant, and the
  proposition that a manifold has Kervaire invariant one.  Only the laws needed
  to state and apply Browder's criterion are exposed.
\end{definition}

\begin{definition}[Classical--synthetic sphere coherence]
  \label{def:sphere-adams-coherence}
  \uses{def:stable-context,def:stable-sphere-suspension,def:synthetic-context,def:synthetic-sphere,def:classical-adams-ss,def:synthetic-adams-ss,def:adams-hj,def:bhs-aim-regrading}
  \lean{KIP126.Kervaire.SphereAdamsCoherence}
  \lean{KIP126.Kervaire.SyntheticHomotopyContext}
  \notready
  Fix a stable context $\mathcal C$, a synthetic context $\mathcal S$ over
  $\mathcal C$, a classical Adams sequence $A_{\mathrm{cl}}$ on
  $S^0_{\mathcal C}$, and a synthetic Adams sequence $A_{\mathrm{syn}}$ on
  $\nu_{\mathcal S}(S^0_{\mathcal C})=S^{0,0}_{\mathcal S}$.  A sphere
  coherence datum records the object identifications, the common unit and
  sphere multiplication, and the grading and naming transports for the classes
  used below.  It does not assert the classical--synthetic $E_2$ identification
  or differential comparison: those are carried by the separately named
  \texttt{syntheticRigidity} external result.  When an object equality is not
  definitional, its equality or transport is an explicit field.  This datum
  contains no rigidity, differential, convergence, or computation assertion.
\end{definition}

\begin{definition}[Classical interpretation of the computation catalogue]
  \label{def:classical-catalogue-coherence}
  \uses{def:near126-computation-input,def:lin-spectrum-catalogue,def:lin-e2-page-catalogue,def:lin-map-catalogue,def:typed-class-expression,def:appendix-evidence-record,def:stable-context,def:classical-adams-ss,def:sphere-adams-coherence,thm:adams-external-pairing}
  \notready
  Fix a raw near-$126$ computation input $D$, a stable context $\mathcal C$,
  the selected sphere sequence $A_{\mathrm{cl}}$, and its sphere coherence.
  A classical catalogue coherence datum contains:
  \begin{enumerate}
    \item for every spectrum identifier of $D$, an admissible spectrum of
      $\mathcal C$ and a chosen classical Adams sequence, with the sphere
      identifier identified with $S^0_{\mathcal C}$ and its sequence
      transported to exactly $A_{\mathrm{cl}}$;
    \item for every typed class expression, an element of the stated spectrum's
      $E_2$ page in exactly its stored bidegree, preserving zero, sums, named
      generators, external products, and module actions whenever those
      operations are present in the catalogue;
    \item for every map identifier, a stable map with the stored suspension,
      source, and target, together with its induced page map; identities,
      compositions, cofiber keys, and the stored linear $E_2$ action commute
      with these interpretations;
    \item a typing equality saying that every appendix row, differential key,
      product record, and finite negative query in $D$ states its proposition
      about these interpreted pages and elements.
  \end{enumerate}
  These are structure fields, not a blanket assertion named
  ``catalogue models the sphere.''  In particular, they expose the transports
  used when a table theorem is applied to $A_{\mathrm{cl}}$.
\end{definition}
