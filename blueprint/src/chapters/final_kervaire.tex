\chapter{Geometric conclusions}
\label{chap:final-kervaire}

This chapter derives the conditional dimension-$126$ manifold and the final dimension list from the internal permanent-cycle theorem and the explicitly listed geometric inputs.

\section{Classical two-line consequence}

% SOURCE: aimpaper/main.tex:151-163, Corollary cor:2line and its cited inputs.
\begin{corollary}[Survivors on the Adams two-line]
  \label{cor:adams-two-line-survivors}
  \uses{def:adams-hj,thm:external-adams-one-line,thm:h6-square-permanent,def:external-result,def:external-evidence}
  \notready
  Subject to the explicitly cited Ext calculations and differential results,
  the nonzero classes on the Adams two-line that survive to $E_\infty$ are
  precisely
  \[
    h_0h_2,\quad h_0h_3,\quad h_2h_4,\quad
    h_1h_j\ (j\ge3),\quad h_j^2\ (0\le j\le6).
  \]
\end{corollary}

\begin{proof}
  The supplied Adams, Lin, May, Mahowald--Tangora, Isaksen--Wang--Xu,
  Mahowald, Barratt--Jones--Mahowald, and Hill--Hopkins--Ravenel results first
  enumerate the $E_3$ candidates and then remove $h_2h_5$, $h_3h_5$,
  $h_3h_6$, and every $h_j^2$ with $j\ge7$.  Their positive survival results
  give every listed class except $h_6^2$; apply
  Theorem~\ref{thm:h6-square-permanent} for that final class.  The dependency
  points from this corollary to the main theorem, never conversely.
\end{proof}

\section{Conditional geometric endpoints}

% SOURCE: aimpaper/main.tex:102-104, Theorem thm:main; made conditional by PROJECT_BOUNDARY.md.
\begin{theorem}[Conditional Kervaire manifold in dimension $126$]
  \label{thm:kervaire-126-conditional}
  \uses{def:kervaire-main-input,thm:h6-square-permanent,thm:external-browder-criterion,def:kip126-trust-boundary}
  \notready
  For every main input $I$, there exists a closed framed smooth
  $126$-manifold in $I$'s geometric context with Kervaire invariant one.
\end{theorem}

\begin{proof}
  Apply Theorem~\ref{thm:h6-square-permanent} to $I$, obtaining survival of
  $h_6^2$, whose stem is $2^7-2=126$.  Apply the forward implication of
  $I.\mathrm{geometry}.\mathrm{browder}$; by construction its permanent-cycle
  premise is stated in $I.\mathrm{classicalSphere}$, the same sequence used by
  the first step.  Every high-stem computation used there is a projection of
  $I.\mathrm{computation}$ and its coherence witnesses.
\end{proof}

% SOURCE: aimpaper/main.tex:108-110, unlabelled Corollary 1.2; made conditional by PROJECT_BOUNDARY.md.
\begin{corollary}[Conditional exact list of Kervaire dimensions]
  \label{cor:kervaire-dimensions-exact}
  \uses{def:kervaire-main-input,thm:kervaire-126-conditional,thm:external-low-kervaire-existence,thm:external-hhr-nonexistence,thm:external-browder-criterion}
  \notready
  For every main input $I$ and natural number $n$, a framed smooth manifold in
  $I.\mathrm{geometry}.\mathrm{framed}$ with Kervaire invariant one exists in
  dimension $n$ if and only if
  \[
    n=2\ \vee\ n=6\ \vee\ n=14\ \vee\ n=30\ \vee\ n=62\ \vee\ n=126.
  \]
\end{corollary}

\begin{proof}
  The low-dimensional fields of $I.\mathrm{geometry}$ and the conditional
  $126$ theorem give existence in every listed dimension.  The Browder field
  restricts any other positive dimension to $2^{j+1}-2$; the low-index fields
  cover $1\le j\le5$, the new theorem covers $j=6$, and the HHR field excludes
  every $j\ge7$.
\end{proof}
